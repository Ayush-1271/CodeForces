import sys
def solve():
    input = sys.stdin.readline
    t = int(input())
    
    for _ in range(t):
        n  = int(input())
        s = input().strip()
        
        cur = 1
        ans = 1
 
        for i in range(1, n):
            if s[i] == s[i - 1]:
                cur += 1
            else:
                cur = 1
 
            ans = max(ans, cur)
 
        print(ans + 1)
        
solve()

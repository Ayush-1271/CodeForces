import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        x, k = map(int, input().split())
        
        if x % k != 0:
            print(1)
            print(x)
        else:
            if k % 2 == 0:
                print(2)
                print(x-3, 3)
            else:
                print(2)
                print(x-2, 2)
            
solve()

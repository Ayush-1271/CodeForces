import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    
    for _ in range(t):
        n, m = map(int, input().split())
        x = input().strip()
        s = input().strip()
        c = 0
        while s not in x and len(x) < n+m:
            x += x
            c += 1
        if s in x:
            print(c)
        else:
            print(-1)
solve()

import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        n, k, x = map(int, input().split())
        
        if x != 1:
            print("YES")
            print(n)
            print(*[1]*n)
        elif x == 1 and k == 1:
            print("NO")
        elif x == 1 and k>1 and n%2 == 0:
            print("YES")
            print(n//2)
            print(*[2]*(n//2))
        elif x == 1 and k>2 and n%2 != 0:
            print("YES")
            print((n-3)//2 +1)
            print(*[2] * ((n - 3) // 2) + [3])
        else:
            print("NO")
            
solve()

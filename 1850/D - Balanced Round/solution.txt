import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
 
    for _ in range(t):
        n, k = map(int, input().split())
        a = list(map(int, input().split()))
 
        a.sort()
 
        longest = 1
        cur = 1
 
        for i in range(1, n):
            if a[i] - a[i - 1] <= k:
                cur += 1
            else:
                cur = 1
 
            longest = max(longest, cur)
 
        print(n - longest)
 
solve()

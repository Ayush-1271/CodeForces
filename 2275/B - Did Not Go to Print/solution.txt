import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    
    for _ in range(t):
        n = int(input())
        s = input().strip()
        
        p = []
        o = [False] * (n+1)
        
        for i, v in enumerate(s, 1):
            if v == '1':
                p.append(i)
            elif v == '2':
                if p:
                    o[p.pop()] = True
                else:
                    o[i] = True
            else:
                o[i] = True
        
        
        ans = []
        for i in range(1, n+1):
            if not o[i]:
                ans.append(i)
        
        print(len(ans))
        if len(ans) > 0:
            print(*ans)
        else:
            print()
        
solve()

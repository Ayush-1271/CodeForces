import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        n, k = map(int, input().split())
        a = list(map(int, input().split()))
        
        m = max(a[0], (k-a[-1])*2)
        for i in range(1, n):
            m = max(m, (a[i]-a[i-1]))
        print(m)
solve()

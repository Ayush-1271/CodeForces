import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
 
    for _ in range(t):
        n = int(input())
        a = sorted(map(int, input().split()))
 
        mc, c = 1, 1
        for i in range(1, n):
            if a[i] == a[i - 1]:
                c += 1
            else:
                mc = max(mc, c)
                c = 1
        mc = max(mc, c)
        
        op = 0
        
        while mc < n:
            op += 1
            if mc * 2 <= n:
                op += mc
                mc *= 2
            else:
                op += n - mc
                mc = n
 
        print(op)
 
solve()

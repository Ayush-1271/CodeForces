import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        
        if len(set(a)) == 1:
            print(-1)
        else:
            b , c = [], []
            m = max(a)
            for i in a:
                if i != m:
                    b.append(i)
                else:
                    c.append(i)
            print(len(b), len(c))
            print(*b)
            print(*c)
            
solve()

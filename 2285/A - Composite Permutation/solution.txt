import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
 
    for _ in range(t):
        n = int(input())
 
        if n < 4:
            print(-1)
            continue
 
        ans = [4]
 
        for i in range(n, 1, -1):
            if i != 4:
                ans.append(i)
 
        ans.insert(3, 1)
 
        print(*ans)
 
solve()

import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        n  = int(input())
        a = list(map(int, input().split()))
        p = 1
        for i in a:
            p *= i
        q = 1
        for i in range(n):
            q *= a[i]
            p //= a[i]
            if p == q:
                print(i+1)
                break
        else:
            print(-1)
solve()

import sys
 
input = sys.stdin.read
 
data = input().split()
index = 0
 
t = int(data[index])
index += 1
 
for _ in range(t):
    n = int(data[index])
    k = int(data[index + 1])
    q = int(data[index + 2])
    index += 3
 
    a = []
    for i in range(n):
        temp = int(data[index + i])
        a.append(1 if temp <= q else 0)
    index += n
 
    count_of_1s = 0
    ways = 0
 
    for i in range(n):
        if a[i] == 1:
            count_of_1s += 1
        else:
            if count_of_1s >= k:
                diff = count_of_1s - k + 1
                ways += (diff * (diff + 1)) // 2
            count_of_1s = 0
 
    if count_of_1s >= k:
        diff = count_of_1s - k + 1
        ways += (diff * (diff + 1)) // 2
 
    print(ways)

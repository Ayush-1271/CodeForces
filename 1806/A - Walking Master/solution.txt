import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        a,b, c,d = map(int, input().split())
        
        di = d-b
        if di < 0:
            print(-1)
            continue
        
        le = a+di - c
        if le < 0:
            print(-1)
            continue
        print(di+le)
solve()

import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        if a != sorted(a):
            print(0)
            continue
        m = a[1] - a[0]
        for i in range(1, n):
            m = min(m, a[i] - a[i-1])
        
        print(m//2 + 1)
         
solve()

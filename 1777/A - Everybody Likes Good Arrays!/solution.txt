import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        
        p = 0 if a[0]%2 == 0 else 1
        c = 0
        
        for i in range(1, n):
            if a[i]%2 != p:
                p = a[i]%2
            else:
                p = (a[i]*a[i-1])%2
                c += 1
        print(c)
solve()

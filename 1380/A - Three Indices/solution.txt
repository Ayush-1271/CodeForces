import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
 
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
 
        f = False
 
        for j in range(1, n - 1):
            if a[j] > a[j - 1] and a[j] > a[j + 1]:
                print("YES")
                print(j, j + 1, j + 2)
                f = True
                break
 
        if not f:
            print("NO")
 
solve()

import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = input().strip()
        l, r = 0, n-1
        c = n
        while l<r:
            if a[l] == a[r]:
                break
            l += 1
            c -=2
            r -= 1
        print(c)
solve()

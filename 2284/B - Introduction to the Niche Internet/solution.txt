import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        n, k = map(int, input().split())
        a = list(map(int, input().split()))
        a.sort()
        
        m = a[n-k:] + a[:n-k][::-1]
        
        t = m[0]
        
        p = 1
        
        for i in range(1, n):
            if m[i]*i >= t:
                p += 1
            t += m[i]
        
        if p == k:
            print("YES")
            print(*m)
        else:
            print("NO")
 
solve()

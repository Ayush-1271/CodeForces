import sys
 
def gcd(a, b):
    while b:
        a, b = b, a%b
    return a
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        p = False
        for i in range(n):
            for j in range(i+1, n):
                if gcd(a[i], a[j]) <= 2:
                    
                    p = True
                    break
            if p:
                break
        if p:
            print("Yes")
        else:
            print("No")
solve()

import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    
    for _ in range(t):
        n, q = map(int, input().split())
        a = list(map(int, input().split()))
        
        p = [0] * (n + 1)
        for i in range(n):
            p[i + 1] = p[i] + a[i]
        
        s = p[n]
        
        for __ in range(q):
            l, r, k = map(int, input().split())
            s1 = p[r] - p[l - 1]
            if (s - s1 + k * (r - l + 1)) % 2 == 1:
                print("YES")
            else:
                print("NO")
solve()

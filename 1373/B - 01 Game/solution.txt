import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        s = input().strip()
        
        o = s.count('1')
        z = s.count('0')
        m = min(o, z)
        
        if m%2 == 0:
            print("NET")
        else:
            print("DA")
        
solve()

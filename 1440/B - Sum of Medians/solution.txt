import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
 
    for _ in range(t):
        n, k = map(int, input().split())
        a = list(map(int, input().split()))
 
        need = n // 2
        idx = n * k - need - 1
 
        s = 0
 
        for i in range(k):
            s += a[idx]
            idx -= need + 1
 
        print(s)
 
solve()

import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        n = int(input())
        
        if n == 1:
            print(0)
            continue
        f = True
        c = 0
        while n != 1 and f:
            if n%6 != 0:
                n *= 2
                if n%6 != 0:
                    f = False
                c += 1
            else:
                n //= 6
                c += 1
        if not f:
            print(-1)
        else:
            print(c)
solve()

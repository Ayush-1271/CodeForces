import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    
    for _ in range(t):
        n , k = map(int, input().split())
        
        if n%2 == 0:
            print("YES")
        else:
            if k%2 == 0:
                print("NO")
            else:
                print("YES")
 
solve()

import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for t in range(t):
        n, k = map(int, input().split())
        a = list(map(int, input().split()))
        
        f = [0]*(n+1)
        for i in a:
            f[i] += 1
 
        s, m = 0, 0   
        for i in range(n+1):
            s += f[i]
            if f[i] == 0 or s < k*(i+1):
                break
            m = i + 1
        
        print(str(m))
solve()

import sys
def solve():
    input = sys.stdin.readline
    t = int(input())
    
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        o, z = 0, 0
        for i in a:
            if i == 1:
                o += 1
            if i == 0:
                z += 1
        print(o * (2**z))
solve()

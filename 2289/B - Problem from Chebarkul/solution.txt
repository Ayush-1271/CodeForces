import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    for _ in range(t):
        n, k = map(int, input().split())
        a = list(map(int, input().split()))
        
        p = [0] * n
        
        for i in range(1, n):
            p[i] = p[i-1] + (a[i] != a[i-1])
        
        ans = 0
        
        for i in range(k):
            l = i
            r = n -k + i
            if p[r] - p[l] == 0:
                ans += 1
        
        print(ans)
solve()

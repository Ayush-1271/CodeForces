import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
    
    for _ in range(t):
        
        n, p = map(int, input().split())
        a = list(map(int, input().split()))
        b = list(map(int, input().split()))
        
        people = []
        
        for i in range(n):
            people.append((b[i], a[i]))
        
        people.sort()
        
        ans = p
        rem = n-1
        
        for i, j in people:
            if i >= p:
                break
            use = min(j, rem)
            
            ans += use * i
            rem -= use
            
            if rem == 0:
                break
        ans += rem * p
        print(ans)
solve()

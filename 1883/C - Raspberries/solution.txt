import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
 
    for _ in range(t):
        n, k = map(int, input().split())
        a = list(map(int, input().split()))
 
        if any(i % k == 0 for i in a):
            print(0)
            continue
 
        if k == 4:
            c = 0
            m = float('inf')
 
            for i in a:
                if i % 2 == 0:
                    c += 1
 
                m = min(m, (4 - i % 4) % 4)
 
            if c >= 2:
                print(0)
            elif c == 1:
                print(1)
            else:
                print(min(m, 2))
 
        else:
            m = float('inf')
 
            for i in a:
                m = min(m, (k - i % k) % k)
 
            print(m)
 
solve()

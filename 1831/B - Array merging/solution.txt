import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
 
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        b = list(map(int, input().split()))
 
        freq_a = {}
        freq_b = {}
 
        c_a, c_b = 1, 1
 
        for i in range(1, n):
            if a[i] == a[i - 1]:
                c_a += 1
            else:
                x = a[i - 1]
                freq_a[x] = max(freq_a.get(x, 0), c_a)
                c_a = 1
 
            if b[i] == b[i - 1]:
                c_b += 1
            else:
                x = b[i - 1]
                freq_b[x] = max(freq_b.get(x, 0), c_b)
                c_b = 1
 
        freq_a[a[-1]] = max(freq_a.get(a[-1], 0), c_a)
        freq_b[b[-1]] = max(freq_b.get(b[-1], 0), c_b)
 
        ans = 0
        for x in freq_a.keys() | freq_b.keys():
            ans = max(ans, freq_a.get(x, 0) + freq_b.get(x, 0))
 
        print(ans)
 
solve()

import sys
 
def solve():
    input = sys.stdin.readline
    t = int(input())
 
    for _ in range(t):
        n = int(input())
        v = list(map(int, input().split()))
 
        ans = v[-1] - v[0]
 
        for i in range(n):
            ans = max(ans, v[i] - v[0])
            ans = max(ans, v[-1] - v[i])
 
        for i in range(n - 1):
            ans = max(ans, v[i] - v[i + 1])
 
        print(ans)
 
solve()
